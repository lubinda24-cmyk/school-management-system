function getGrade(mark) {
  if (mark === null || mark === undefined || isNaN(mark)) return "-";
  if (mark >= 75) return "1";
  if (mark >= 70) return "2";
  if (mark >= 65) return "3";
  if (mark >= 60) return "4";
  if (mark >= 55) return "5";
  if (mark >= 50) return "6";
  if (mark >= 45) return "7";
  if (mark >= 40) return "8";
  return "9";
}

function gradeColor(grade) {
  const g = parseInt(grade);
  if (isNaN(g)) return "#64748b";
  if (g <= 2) return "#16a34a";
  if (g <= 4) return "#2563eb";
  if (g <= 6) return "#d97706";
  if (g <= 8) return "#dc2626";
  return "#7c3aed";
}

function calcRow(row) {
  const pre = parseFloat(row.preMid);
  const mid = parseFloat(row.midterm);
  const exam = parseFloat(row.exam);
  const ca = (!isNaN(pre) && !isNaN(mid)) ? (pre + mid) / 2 : null;
  const ca40 = ca !== null ? ca * 0.4 : null;
  const exam60 = !isNaN(exam) ? exam * 0.6 : null;
  const overall = (ca40 !== null && exam60 !== null) ? ca40 + exam60 : null;
  const grade = overall !== null ? getGrade(overall) : "-";
  return { ca, ca40, exam60, overall, grade };
}

    const existing = marks[key] || [];
    // generate 30 pupils if none
    if (existing.length === 0) {
      const pupils = Array.from({length:30},(_,i)=>({
        id:i+1, name:`Pupil ${i+1}`, preMid:"", midterm:"", exam:""
      }));
      setRows(pupils);
    } else {
      setRows(JSON.parse(JSON.stringify(existing)));
    }
    setLoaded(true);
  }

  function updateRow(idx, field, val) {
    setRows(prev => {
      const next = [...prev];
      next[idx] = {...next[idx], [field]: val};
      return next;
    });
  }

  function handleSave() {
    if (!key) return;
    setMarks(prev => {
      const next = {...prev, [key]: rows};
      saveMarks(next);
      return next;
    });
    showToast("✅ Marks saved successfully!");
  }

  const classOptions = user.role === "admin" ? CLASSES : CLASSES;

  return (
    <div>
      <div className="page-header">
        <h1>✏️ Enter Marks</h1>
        <p>Select a class and subject to enter or update marks</p>
      </div>
      <div className="card">
        <div className="filter-row">
          <div className="filter-group">
            <label>Class</label>
            <select value={cls} onChange={e=>{setCls(e.target.value);setLoaded(false)}}>
              <option value="">-- Select Class --</option>
              {classOptions.map(c=><option key={c}>{c}</option>)}
            </select>
          </div>
          <div className="filter-group">
            <label>Subject</label>
            <select value={subject} onChange={e=>{setSubject(e.target.value);setLoaded(false)}}>
              <option value="">-- Select Subject --</option>
              {SUBJECTS.map(s=><option key={s}>{s}</option>)}
            </select>
          </div>
          <button className="btn-secondary" onClick={loadData} disabled={!cls||!subject}>
            📂 Load / Open
          </button>
          {loaded && <button className="btn-save" onClick={handleSave}>💾 Save Marks</button>}
        </div>

        {!loaded && (
          <div className="empty-state">
            <div className="icon">📋</div>
            <p>Select a class and subject, then click <strong>Load / Open</strong></p>
          </div>
        )}

        {loaded && (
          <>
            <div style={{marginBottom:"12px",display:"flex",alignItems:"center",gap:"12px"}}>
              <span className="badge badge-blue">{cls}</span>
              <span className="badge badge-green">{subject}</span>
              <span style={{fontSize:"13px",color:"#64748b"}}>{rows.length} pupils</span>
            </div>
            <div className="table-wrap">
              <table>
                <thead>
                  <tr>
                    <th>#</th>
                    <th>Pupil Name</th>
                    <th>Pre-Midterm<br/><span style={{fontWeight:400,fontSize:11}}>out of 100</span></th>
                    <th>Midterm<br/><span style={{fontWeight:400,fontSize:11}}>out of 100</span></th>
                    <th style={{background:"#1e4a8a"}}>CA Mark</th>
                    <th style={{background:"#1e4a8a"}}>40% of CA</th>
                    <th>Exam Mark<br/><span style={{fontWeight:400,fontSize:11}}>out of 100</span></th>
                    <th style={{background:"#1e4a8a"}}>60% of Exam</th>
                    <th style={{background:"#0f2d5e"}}>Overall %</th>
                    <th style={{background:"#0f2d5e"}}>Grade</th>
                  </tr>
                </thead>
                <tbody>
                  {rows.map((row,idx)=>{
                    const {ca,ca40,exam60,overall,grade} = calcRow(row);
                    const fmt = v => v!==null ? v.toFixed(1) : "-";
                    return (
                      <tr key={row.id}>
                        <td style={{color:"#94a3b8",fontSize:12}}>{idx+1}</td>
                        <td>
                          <input style={{border:"none",background:"transparent",width:"140px",
                            fontFamily:"'DM Sans',sans-serif",fontSize:13,color:"#0f2d5e",fontWeight:500}}
                            value={row.name}
                            onChange={e=>updateRow(idx,"name",e.target.value)} />
                        </td>
                        <td><input className="mark-input" type="number" min="0" max="100"
                          value={row.preMid} onChange={e=>updateRow(idx,"preMid",e.target.value)} /></td>
                        <td><input className="mark-input" type="number" min="0" max="100"
                          value={row.midterm} onChange={e=>updateRow(idx,"midterm",e.target.value)} /></td>
                        <td className="calc-cell" style={{background:"#f0f7ff"}}>{fmt(ca)}</td>
                        <td className="calc-cell" style={{background:"#f0f7ff"}}>{fmt(ca40)}</td>
                        <td><input className="mark-input" type="number" min="0" max="100"
                          value={row.exam} onChange={e=>updateRow(idx,"exam",e.target.value)} /></td>
                        <td className="calc-cell" style={{background:"#f0f7ff"}}>{fmt(exam60)}</td>
                        <td className="calc-cell" style={{background:"#e8f0fe",fontWeight:700,fontSize:15}}>
                          {overall!==null ? overall.toFixed(1) : "-"}
                        </td>
                        <td style={{background:"#e8f0fe"}}>
                          {grade!=="-" ? (
                            <div className="grade-badge" style={{background:gradeColor(grade),margin:"0 auto"}}>
                              {grade}
                            </div>
                          ) : "-"}
                        </td>
                      </tr>
                    );
                  })}
                </tbody>
              </table>
            </div>
            <div style={{marginTop:"16px",display:"flex",justifyContent:"flex-end"}}>
              <button className="btn-save" onClick={handleSave}>💾 Save All Marks</button>
            </div>
          </>
        )}
      </div>
    </div>
  );
}

function ResultsPage({ marks }) {
  const [filterCls, setFilterCls] = useState("");
  const [filterSubject, setFilterSubject] = useState("");

  const key = filterCls && filterSubject ? `${filterCls}__${filterSubject}` : null;
  const rows = key ? (marks[key] || []) : [];
  const withData = rows.filter(r => calcRow(r).overall !== null);

  const gradeCounts = {};
  for(let i=1;i<=9;i++) gradeCounts[i]=0;
  withData.forEach(r => {
    const g = parseInt(calcRow(r).grade);
    if (!isNaN(g)) gradeCounts[g]=(gradeCounts[g]||0)+1;
  });

  const avg = withData.length > 0
    ? withData.reduce((s,r) => s + calcRow(r).overall,0) / withData.length
    : null;

  return (
    <div>
      <div className="page-header">
        <h1>📋 Results & Reports</h1>
        <p>View calculated results for any class and subject</p>
      </div>
      <div className="card">
        <div className="filter-row">
          <div className="filter-group">
            <label>Class</label>
            <select value={filterCls} onChange={e=>setFilterCls(e.target.value)}>
              <option value="">-- All Classes --</option>
              {CLASSES.map(c=><option key={c}>{c}</option>)}
            </select>
          </div>
          <div className="filter-group">
            <label>Subject</label>
            <select value={filterSubject} onChange={e=>setFilterSubject(e.target.value)}>
              <option value="">-- All Subjects --</option>
              {SUBJECTS.map(s=><option key={s}>{s}</option>)}
            </select>
          </div>
        </div>

        {key && withData.length > 0 && (
          <>
            <div style={{display:"flex",gap:"16px",flexWrap:"wrap",marginBottom:"20px"}}>
              <div className="stat-card" style={{flex:"0 0 auto",minWidth:"160px",borderLeftColor:"#2563eb"}}>
                <div className="label">Class Average</div>
                <div className="value">{avg.toFixed(1)}%</div>
                <div className="sub">Grade {getGrade(avg)}</div>
              </div>
              <div className="stat-card" style={{flex:"0 0 auto",minWidth:"160px",borderLeftColor:"#16a34a"}}>
                <div className="label">Pass Rate</div>
                <div className="value">{Math.round(withData.filter(r=>calcRow(r).overall>=40).length/withData.length*100)}%</div>
                <div className="sub">{withData.filter(r=>calcRow(r).overall>=40).length} of {withData.length}</div>
              </div>
              <div className="stat-card" style={{flex:"0 0 auto",minWidth:"160px",borderLeftColor:"#f59e0b"}}>
                <div className="label">Pupils Assessed</div>
                <div className="value">{withData.length}</div>
                <div className="sub">With complete marks</div>
              </div>
            </div>

            <div style={{marginBottom:"16px"}}>
              <div className="card-title">Grade Distribution</div>
              <div style={{display:"flex",flexWrap:"wrap",gap:"8px"}}>
                {[1,2,3,4,5,6,7,8,9].map(g => (
                  <div key={g} style={{background:gradeColor(g)+"22",border:`2px solid ${gradeColor(g)}`,
                    borderRadius:"8px",padding:"8px 14px",textAlign:"center",minWidth:"70px"}}>
                    <div style={{fontWeight:700,fontSize:"18px",color:gradeColor(g)}}>Grade {g}</div>
                    <div style={{fontSize:"20px",fontWeight:700,color:"#0f2d5e"}}>{gradeCounts[g]}</div>
                    <div style={{fontSize:"11px",color:"#64748b"}}>pupils</div>
                  </div>
                ))}
              </div>
            </div>
          </>
        )}

        {key && rows.length === 0 && (
          <div className="empty-state">
            <div className="icon">📭</div>
            <p>No marks entered yet for <strong>{filterCls} – {filterSubject}</strong></p>
          </div>
        )}

        {key && rows.length > 0 && (
          <div className="table-wrap">
            <table>
              <thead>
                <tr>
                  <th>#</th>
                  <th>Pupil Name</th>
                  <th>Pre-Mid</th>
                  <th>Midterm</th>
                  <th>CA</th>
                  <th>40% CA</th>
                  <th>Exam</th>
                  <th>60% Exam</th>
                  <th>Overall %</th>
                  <th>Grade</th>
                </tr>
              </thead>
              <tbody>
                {rows.map((row,idx) => {
                  const {ca,ca40,exam60,overall,grade} = calcRow(row);
                  const fmt = v => v!==null ? v.toFixed(1) : "-";
                  return (
                    <tr key={idx}>
                      <td style={{color:"#94a3b8",fontSize:12}}>{idx+1}</td>
                      <td style={{fontWeight:500}}>{row.name}</td>
                      <td>{row.preMid||"-"}</td>
                      <td>{row.midterm||"-"}</td>
                      <td className="calc-cell">{fmt(ca)}</td>
                      <td className="calc-cell">{fmt(ca40)}</td>
                      <td>{row.exam||"-"}</td>
                      <td className="calc-cell">{fmt(exam60)}</td>
                      <td style={{fontWeight:700,color:"#0f2d5e"}}>{overall!==null?overall.toFixed(1):"-"}</td>
                      <td>
                        {grade!=="-" ? (
                          <div className="grade-badge" style={{background:gradeColor(grade),margin:"0 auto"}}>
                            {grade}
                          </div>
                        ) : "-"}
                      </td>
                    </tr>
                  );
                })}
              </tbody>
            </table>
          </div>
        )}

        {!key && (
          <div className="empty-state">
            <div className="icon">🔍</div>
            <p>Select a class and subject to view results</p>
          </div>
        )}
      </div>
    </div>
  );
}

// ── MAIN APP ─────────────────────────────────────────────────────────────────
export default function App() {
  const [user, setUser] = useState(null);
  const [page, setPage] = useState("dashboard");
  const [marks, setMarks] = useState(loadMarks);
  const [toast, setToast] = useState("");
  const [sidebarOpen, setSidebarOpen] = useState(false);

  function showToast(msg) {
    setToast(msg);
    setTimeout(()=>setToast(""),3000);
  }

  if (!user) return (
    <>
      <style>{css}</style>
      <LoginPage onLogin={u=>{setUser(u);setPage("dashboard")}} />
    </>
  );

  return (
    <>
      <style>{css}</style>
      <div className="app-layout">
        <Sidebar user={user} page={page} setPage={setPage}
          onLogout={()=>{setUser(null);setPage("dashboard")}}
          open={sidebarOpen} setOpen={setSidebarOpen} />
        <main className="main-content">
          {page==="dashboard" && <Dashboard user={user} marks={marks} />}
          {page==="marks" && <MarksPage user={user} marks={marks} setMarks={setMarks} showToast={showToast} />}
          {page==="results" && <ResultsPage marks={marks} />}
          {page==="teachers" && user.role==="admin" && <TeachersPage />}
          {page==="classes" && user.role==="admin" && <ClassesPage />}
        </main>
      </div>
      {toast && <div className="success-toast">{toast}</div>}
    </>
  );
}
